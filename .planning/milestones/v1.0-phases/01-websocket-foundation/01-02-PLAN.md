       def run(self):
           self._running = True
           while self._running:
               self._ws_app = websocket.WebSocketApp(
                   self.WS_URL,
                   on_message=self._on_message,
                   on_error=self._on_error,
                   on_close=self._on_close,
                   on_open=self._on_open,
               )
               t = threading.Thread(
                   target=self._ws_app.run_forever,
                   kwargs={"ping_interval": 0, "ping_timeout": None},
                   daemon=True,
               )
               t.start()
               self._connection_started_at = time.monotonic()
               t.join()  # blocks until WS thread exits
               if not self._running:
                   break

               if time.monotonic() - self._connection_started_at > 60:
                   self._reconnect_attempt = 0
               delay = reconnect_delay(self._reconnect_attempt)
               self._reconnect_attempt += 1
               print(f"[sports_ws] event=reconnect_scheduled delay={delay:.1f}s attempt={self._reconnect_attempt}")
               time.sleep(delay)

       def _purge_ended_games(self):
